Google Drive sheet setup

Connect your private Google Sheet. No API keys or cloud billing.

1
Copy the script
Copy the script below, then paste it into the Code.gs editor in Google Apps Script.
2
Create and authorize your sheet
In the toolbar beside Debug, select createMySheetNow and click Run. When prompted, choose Review permissions > Advanced > Go to QR Scanner (unsafe) > Allow. Your sheet is created in Google Drive.
3
Deploy and connect
Choose Deploy > Manage deployments (or New deployment > Web app). Select the pencil and set Version to New version. Click Deploy, then copy the Web app URL ending in /exec and paste it into the webhook field.
Set “Who has access” to “Anyone” to allow extension sync.
Universal Apps Script Code.gs
// STEP 1: Create your sheet and authorize Google function createMySheetNow() { var ss = getOrCreateSheet(); Logger.log("SUCCESS! Your sheet is ready at: " + ss.getUrl()); return ss.getUrl(); } // STEP 2: Automatic webhook handlers for extension scans function doPost(e) { return handleScan(e); } function doGet(e) { if (e && e.parameter && (e.parameter.text || e.parameter.format)) { return handleScan(e); } return ContentService.createTextOutput("QR Scanner Cloud Webhook is Active!").setMimeType(ContentService.MimeType.TEXT); } function handleScan(e) { try { var ss = getOrCreateSheet(); var sheet = ss.getActiveSheet(); var data = {}; if (e && e.postData && e.postData.contents) { try { data = JSON.parse(e.postData.contents); } catch (pErr) {} } if (!data.text && e && e.parameter) { data = e.parameter; } sheet.appendRow([new Date(), data.format || "Barcode", data.text || ""]); return ContentService.createTextOutput("OK").setMimeType(ContentService.MimeType.TEXT); } catch (err) { return ContentService.createTextOutput("Error: " + err.message).setMimeType(ContentService.MimeType.TEXT); } } // Helper: gets active sheet if bound, or creates/finds QR_Barcode_Scans in Google Drive root function getOrCreateSheet() { var ss; try { ss = SpreadsheetApp.getActiveSpreadsheet(); } catch (err) {} if (!ss) { var files = DriveApp.getFilesByName("QR_Barcode_Scans"); if (files.hasNext()) { ss = SpreadsheetApp.open(files.next()); } else { ss = SpreadsheetApp.create("QR_Barcode_Scans"); var s = ss.getActiveSheet(); s.appendRow(["Timestamp", "Format", "Scanned Content"]); s.getRange(1, 1, 1, 3).setFontWeight("bold").setBackground("#3b82f6").setFontColor("#ffffff"); } } return ss; }
